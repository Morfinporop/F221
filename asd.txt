local function LoadCfg()
    if not file.Exists(CfgFile, "DATA") then return end
    local t = util.JSONToTable(file.Read(CfgFile, "DATA") or "")
    if not istable(t) then return end
    for k, v in pairs(Defaults) do
        local n = t[k]
        if istable(v) then
            if istable(n) then
                local c = {}
                for i = 1, 3 do
                    c[i] = math.Clamp(tonumber(n[i]) or v[i], 0, 255)
                end
                Cfg[k] = c
            end
        elseif type(n) == type(v) then
            Cfg[k] = n
        end
    end
    if t.glowSettingRevision ~= GlowCfgRevision then
        Cfg.glowPower = Defaults.glowPower
        migratedGlow = true
    end
    Cfg.scale = math.Clamp(Cfg.scale, 0.7, 1)
    Cfg.radius = math.Clamp(Cfg.radius, 0, 30)
    Cfg.bgAlpha = math.Clamp(Cfg.bgAlpha, 0, 255)
    Cfg.cardAlpha = math.Clamp(Cfg.cardAlpha, 0, 255)
    Cfg.glowPower = math.Clamp(Cfg.glowPower, 0.2, 5)
end

LoadCfg()

    end)
end

local PolyCache = {}
local PolyCount = 0

local function RBox(r, x, y, w, h, col)
    w = math.Round(w)
    h = math.Round(h)
    if w <= 0 or h <= 0 then return end
    local a = math.Clamp(col.a or 255, 0, 255)
    if a <= 0 then return end
    draw.NoTexture()
    surface.SetDrawColor(col.r, col.g, col.b, a)
    r = math.min(math.Round(r), math.floor(w / 2), math.floor(h / 2))
    if r <= 0 then
        surface.DrawRect(x, y, w, h)
        return
    end
    local key = w .. ":" .. h .. ":" .. r
    local e = PolyCache[key]
    if not e then
        if PolyCount > 800 then
            PolyCache = {}
            PolyCount = 0
        end
        local seg = math.Clamp(math.ceil(r / 2), 4, 20)
        local corners = {
            {r, r, 180},
            {w - r, r, 270},
            {w - r, h - r, 0},
            {r, h - r, 90}
        }
        local base, out = {}, {}
        for _, c in ipairs(corners) do
            for i = 0, seg do
                local ang = math.rad(c[3] + 90 * i / seg)
                base[#base + 1] = {x = c[1] + math.cos(ang) * r, y = c[2] + math.sin(ang) * r}
                out[#out + 1] = {x = 0, y = 0}
            end
        end
        e = {base = base, out = out}
        PolyCache[key] = e
        PolyCount = PolyCount + 1
    end
    for i, v in ipairs(e.base) do
        local o = e.out[i]
        o.x = x + v.x
        o.y = y + v.y
    end
    surface.DrawPoly(e.out)
end

local function Ease(t)
    t = math.Clamp(t, 0, 1)
    return 1 - (1 - t) ^ 3
end

local function EA(x, d)
    return Ease(x / d)
end

local function Approach(cur, target, speed)
    return Lerp(math.min(1, FrameTime() * speed), cur, target)
end

local function AC(a, hv)
    hv = hv or 0
    local c = Cfg.accent
    return Color(Lerp(hv * 0.2, c[1], 255), Lerp(hv * 0.2, c[2], 255), Lerp(hv * 0.2, c[3], 255), a or 255)
end

local function OnAC()
    local c = Cfg.accent
    local lum = c[1] * 0.299 + c[2] * 0.587 + c[3] * 0.114
    if lum > 150 then return Color(17, 17, 17) end
    return Color(255, 255, 255)
end

local function BGC(a)
    local c = Cfg.bg
    return Color(c[1], c[2], c[3], a)
end

local function GetScl()
    return math.min(ScrW() / 1920, ScrH() / 1080) * Cfg.scale
end

local function Fit(txt, font, maxw)
    surface.SetFont(font)
    if surface.GetTextSize(txt) <= maxw then return txt end
    while (utf8.len(txt) or 0) > 1 and surface.GetTextSize(txt .. "…") > maxw do
        txt = string.sub(txt, 1, utf8.offset(txt, -1) - 1)
    end
    return txt .. "…"
end

local function CanSee(v, pl)
    local al = v.allowed
    if isnumber(al) then return al == pl:Team() end
    if istable(al) and #al > 0 then return table.HasValue(al, pl:Team()) end
    return true
end

local function IsLocked(v, pl)
    if not v.customCheck then return false end
    local ok, res = pcall(v.customCheck, pl)
    return ok and not res
end

        if #items > 0 then

                price = v.price or 0,

            }
        end
    end
    add("ПРЕДМЕТЫ", Color(255, 138, 0), ents)

    local guns = {}
    for _, v in ipairs(CustomShipments or {}) do
        if v.separate and v.pricesep and CanSee(v, pl) then
            guns[#guns + 1] = {
                name = v.name,
                model = v.model,
                price = v.pricesep,
                locked = IsLocked(v, pl),
                info = "Одна штука",
                act = function() RunConsoleCommand("DarkRP", "buy", v.name) end
            }
        end
    end
    add("ОРУЖИЕ", Color(47, 107, 255), guns)

    local ships = {}
    for _, v in ipairs(CustomShipments or {}) do
        if not v.noship and CanSee(v, pl) then
            ships[#ships + 1] = {
                name = v.name,
                model = v.model,
                price = v.price or 0,
                locked = IsLocked(v, pl),
                info = "Ящик x" .. (v.amount or 1),
                act = function() RunConsoleCommand("DarkRP", "buyshipment", v.name) end
            }
        end
    end
    add("ЯЩИКИ", Color(60, 200, 110), ships)

    local vehs = {}
    for _, v in ipairs(CustomVehicles or {}) do
        if CanSee(v, pl) then
            vehs[#vehs + 1] = {
                name = v.name,
                model = v.model,
                price = v.price or 0,
                locked = IsLocked(v, pl),
                info = "Транспорт",
                act = function() RunConsoleCommand("DarkRP", "buyvehicle", v.name) end
            }

    end
    add("ТРАНСПОРТ", Color(160, 90, 255), vehs)

    local gm = GAMEMODE or GM
    for key, v in pairs(gm and gm.AmmoTypes or {}) do

            info = v.amountGiven and ("Патронов: " .. v.amountGiven) or nil,
            act = function() RunConsoleCommand("DarkRP", "buyammo", v.ammoType or key) end
        }
    end
    table.sort(ammo, function(a, b) return a.name < b.name end)
    add("ПАТРОНЫ", Color(255, 74, 74), ammo)

    return cats
end

local function SearchLower(text)
    local s = string.lower(text or "")
    s = s:gsub("\208[\144-\175]", function(char)
        local second = string.byte(char, 2)
        if second < 160 then return string.char(208, second + 32) end
        return string.char(209, second - 32)
    end)
    return (s:gsub("\208\129", "\209\145"))
end

local function HasCyrillic(text)
    return isstring(text) and string.find(text, "[\208\209][\128-\191]") ~= nil
end

local function BuildCommandData()
    local result = {}
    if not DarkRP then return result end

    local commands = DarkRP.chatCommands
    if isfunction(DarkRP.getChatCommands) then
        local ok, data = pcall(DarkRP.getChatCommands)
        if ok and istable(data) then commands = data end
    end
    if not istable(commands) then return result end

    local prefix = ChatPrefix()
    local pl = LocalPlayer()
    local generated = {}

    for _, job in ipairs(RPExtraTeams or {}) do
        if isstring(job.command) and job.command ~= "" then
            local command = SearchLower(job.command)
            local name = isstring(job.name) and job.name or "Без названия"
            generated[command] = {"Выбор профессии", "Выбрать профессию «" .. name .. "»."}
            generated["vote" .. command] = {"Голосование за профессию", "Начать голосование за «" .. name .. "»."}
        end
    end
    for _, item in ipairs(DarkRPEntities or {}) do
        if isstring(item.cmd) and item.cmd ~= "" then
            local command = SearchLower(item.cmd)
            local name = isstring(item.name) and item.name or "предмет"
            if not generated[command] then
                generated[command] = {"Покупка предмета", "Купить «" .. name .. "»."}
            end
        end
    end

    for key, entry in pairs(commands) do
        if istable(entry) then
            local name = entry.command or key
            if isstring(name) and name ~= "" then
                local commandKey = SearchLower(name)
                local translation = generated[commandKey] or CommandRu[commandKey] or CommandRu[CommandAliases[commandKey] or ""]
                local description = entry.description
                if isfunction(DarkRP.getChatCommandDescription) then
                    local ok, value = pcall(DarkRP.getChatCommandDescription, name)
                    if ok and isstring(value) then description = value end
                end
                if HasCyrillic(entry.descriptionRu) then
                    description = entry.descriptionRu
                elseif HasCyrillic(entry.description_ru) then
                    description = entry.description_ru
                end
                local original = isstring(description) and description or ""
                original = original:gsub("%s+", " ")
                original = string.Trim(original)
                local title = entry.titleRu or entry.nameRu or entry.title_ru or entry.name_ru
                if not HasCyrillic(title) then
                    title = translation and translation[1] or "Команда сервера"
                end
                if HasCyrillic(original) then
                    description = original
                elseif translation then
                    description = translation[2]
                elseif original ~= "" then
                    description = "Описание на русском пока не добавлено"
                else
                    description = "Описание отсутствует"
                end

                local available = true
                if isfunction(entry.condition) then
                    local ok, allowed = pcall(entry.condition, pl)
                    available = ok and not not allowed
                end

                local command = prefix .. name
                local tooltip = command .. " — " .. description
                if not translation and original ~= "" and not HasCyrillic(original) then
                    tooltip = tooltip .. "\nОписание сервера: " .. original
                end
                result[#result + 1] = {
                    command = command,
                    title = title,
                    description = description,
                    tooltip = tooltip,
                    search = SearchLower(command .. " " .. title .. " " .. description .. " " .. original),
                    available = available
                }
            end
        end
    end

    table.sort(result, function(a, b)
        return string.lower(a.command) < string.lower(b.command)
    end)
    return result
end

local function CloseMenu()
    if not IsValid(Pnl) or Pnl.closing then return end
    Pnl.closing = true

local function TakeJob()
    if not IsValid(Pnl) or Pnl.closing then return end
    local j = RPExtraTeams[activeJob]
    if j and j.command then
        local pl = LocalPlayer()
        local model = istable(j.model) and j.model[activeMdl] or j.model
        if j.team and isstring(model) and DarkRP and isfunction(DarkRP.setPreferredJobModel) then
            DarkRP.setPreferredJobModel(j.team, model)
        end
        local vote = j.vote
        if isfunction(j.RequiresVote) then
            local ok, required = pcall(j.RequiresVote, pl, j.team)
            vote = vote or (ok and required)
        end
        pl:ConCommand("say " .. ChatPrefix() .. (vote and "vote" or "") .. j.command)
    end
    CloseMenu()
end

    end

    local function SmoothScroll(sp)
        sp.target = 0
        sp.OnMouseWheeled = function(s, d)
            local mx = math.max(0, s:GetCanvas():GetTall() - s:GetTall())
            s.target = math.Clamp(s.target - d * S(90), 0, mx)
            return true
        end
        sp.Think = function(s)
            local bar = s:GetVBar()
            local cur = bar:GetScroll()
            if bar.Dragging or math.abs(cur - (s.lastScroll or 0)) > 0.5 then
                s.target = cur
            end
            local mx = math.max(0, s:GetCanvas():GetTall() - s:GetTall())
            s.target = math.Clamp(s.target, 0, mx)
            if math.abs(cur - s.target) > 0.5 then
                bar:SetScroll(Lerp(math.min(1, FrameTime() * 12), cur, s.target))
            elseif cur ~= s.target then
                bar:SetScroll(s.target)
            end

    local function SliderRow(label, get, set, mn, mx, fmt, rel)
        local row = NewRow(96)
        local sl = vgui.Create("DPanel", row)
        sl:SetCursor("hand")
        row.PerformLayout = function(s, w, h)
            sl:SetPos(S(30), S(56))
            sl:SetSize(w - S(60), S(26))
        end
        row.Draw = function(s, w, h)
            draw.SimpleText(label, "E_item", S(30), S(30), Color(255, 255, 255), 0, 1)
            draw.SimpleText(fmt(get()), "E_item", w - S(30), S(30), AC(255), 2, 1)
        end
        sl.Think = function(s)
            if not s.drag then return end
            local lx = s:ScreenToLocal(gui.MouseX(), 0)
            local f = math.Clamp(lx / math.max(1, s:GetWide()), 0, 1)
            set(mn + (mx - mn) * f)
        end
        sl.OnMousePressed = function(s, k)
            if k ~= MOUSE_LEFT then return end
            s.drag = true
            s:MouseCapture(true)
        end
        sl.OnMouseReleased = function(s, k)
            if not s.drag then return end
            s.drag = false
            s:MouseCapture(false)
            SaveCfg()
            if rel then rel() end
        end
        sl.Paint = function(s, w, h)
            s.hv = Approach(s.hv or 0, (s:IsHovered() or s.drag) and 1 or 0, 14)
            local f = math.Clamp((get() - mn) / (mx - mn), 0, 1)
            RBox(512, 0, h / 2 - S(4), w, S(8), Color(255, 255, 255, 30))
            RBox(512, 0, h / 2 - S(4), math.max(S(8), w * f), S(8), AC(255))
            local d = S(22) + S(4) * s.hv
            local kx = math.Clamp(w * f, d / 2, w - d / 2)
            RBox(512, kx - d / 2, h / 2 - d / 2, d, d, Color(255, 255, 255))
        end
    end
